 - ```bash
    service docker restart
   ```

- ```sql
    DROP SCHEMA public CASCADE;
    CREATE SCHEMA public;
  ```

    docker compose -f infra.yaml up -d
  ```

    docker compose -f deployment/infra.yaml up -d
  ```

    docker compose -f deployment/infra.yaml down
  ```
